\subsection{Patterns in the answer keys}

\subsection{Distractors that are almost right}
\begin{itemize}
  \item \textbf{Brackets and dimensions:} \code{[20, 90]} vs \code{[[20, 90]]}, and \code{[67 58]} vs \code{[[67 58]]}. Two index lists give a 1D result; boolean row selection keeps 2D.
  \item \textbf{Small details in the output:} two \code{DatetimeIndex} options may differ only by \code{freq=\textquotesingle{}D\textquotesingle{}}. The correct one has it.
  \item \textbf{Fake methods:}
  \begin{itemize}
    \item lists: \code{list.push()};
    \item DataFrames: \code{df.sort()}, \code{df.top()}, \code{df.read()}, \code{df.save\_pickle()}, \code{df.to\_binary()};
    \item pandas functions: \code{pd.load\_json()}, \code{pd.json\_to\_df()};
    \item invalid calls: \code{shift(window=3)}, \code{resample(\textquotesingle{}monthly\textquotesingle{}).average()}, \code{open(...,\textquotesingle{}rw\textquotesingle{})}, \code{arr.transform(...)}, \code{s.applymap}.
  \end{itemize}
  \item \textbf{Same function, different parameter:}
  \begin{itemize}
    \item \code{sort\_index()} vs \code{sort\_index(axis=1)};
    \item \code{shift(-{}2)} vs \code{shift(-{}2, freq=\textquotesingle{}H\textquotesingle{})};
    \item \code{groupby([\textquotesingle{}A\textquotesingle{},\textquotesingle{}B\textquotesingle{}])} vs \code{groupby([\textquotesingle{}B\textquotesingle{},\textquotesingle{}A\textquotesingle{}])};
    \item \code{cut} vs \code{qcut};
    \item \code{keep=\textquotesingle{}first\textquotesingle{}} vs \code{keep=\textquotesingle{}last\textquotesingle{}};
    \item \code{how=\textquotesingle{}left\textquotesingle{}} vs \code{\textquotesingle{}outer\textquotesingle{}};
    \item \code{axis=0} vs \code{axis=1};
    \item \code{.sum()} vs \code{.mean()}.
  \end{itemize}
  \item \textbf{Counting characters:} upper and lower case are different keys, and spaces count too.
  \item \textbf{Not reassigned:} \code{arr.reshape(...)}, \code{np.sort(arr)}, \code{df.drop(...)} and \code{s.str.upper()} leave the original unchanged unless you reassign the result (or use \code{inplace=True} / the in-place method).
\end{itemize}

\subsection{When two answers both seem right, pick the ``standard'' one}
\begin{itemize}
  \item Monthly mean $\to$ \code{df.resample(\textquotesingle{}M\textquotesingle{}).mean()}.
  \item Binary output layer $\to$ 1 neuron with sigmoid. Input neurons = number of numeric features.
  \item Reading a file $\to$ \code{with open(\textquotedbl{}f\textquotedbl{}, \textquotedbl{}r\textquotedbl{}) as f: content = f.read()}.
  \item Palindrome $\to$ \code{s == s[::-{}1]}.
  \item Loading only a few rows $\to$ \code{pd.read\_csv(..., nrows=5)}.
\end{itemize}

\subsection{Method for code questions}
\begin{enumerate}
  \item Write down each variable's value \textbf{line by line}. Never guess the output.
  \item \textbf{NumPy:} sketch the matrix. \code{axis=0} gives one result per column, \code{axis=1} one result per row. Read shapes from the brackets.
  \item \textbf{try/except:} with no exception, \code{else} runs. \code{finally} \textbf{always} runs.
  \item \textbf{Scope:} \code{global} means the module-level variable. A tuple reassigned inside a function does not change outside it; a list or dict modified inside does.
  \item \textbf{Slicing:} apply chained \code{[ ][ ][::2]} one at a time, left to right. The stop index is excluded (except in \code{loc}).
  \item \textbf{Merge:} write down the keys of each table, then keep the keys that \code{how=} says to keep.
\end{enumerate}

\subsection{Time management}
\begin{itemize}
  \item 3 hours for 40 questions is about \textbf{4 minutes per question}.
  \item First do all the theory questions, which take about 30 seconds each. Then do the code-tracing questions.
  \item A blank answer scores 0, the same as a wrong one, so \textbf{never leave an answer blank}.
\end{itemize}
